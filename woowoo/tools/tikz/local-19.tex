% klikaci plocha pres cely uzel (pruhledny obdelnik velikosti uzlu s odkazem)
\newcommand{\glinksize}[1]{\pgfpointdiff{\pgfpointanchor{#1}{south west}}{\pgfpointanchor{#1}{north east}}%
  \pgfgetlastxy{\glw}{\glh}}
\newcommand{\glink}[2]{\glinksize{#1}%
  \node[inner sep=0pt,outer sep=0pt] at (#1.center) {\hyperlink{#2}{\phantom{\rule{\glw}{\glh}}}};}
\newcommand{\glinkr}[2]{\glinksize{#1}%
  \node[inner sep=0pt,outer sep=0pt] at (#1.center) {\hyperref[#2]{\phantom{\rule{\glw}{\glh}}}};}
